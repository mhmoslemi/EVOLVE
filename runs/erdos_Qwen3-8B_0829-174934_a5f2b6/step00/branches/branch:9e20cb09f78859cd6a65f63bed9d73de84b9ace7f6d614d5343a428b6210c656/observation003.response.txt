import numpy as np
import random
import copy

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Ensure initial sum is correct
    total = sum(initial_h)
    initial_h = [x * (n_points / 2) / total for x in initial_h]
    
    # Function to compute C5
    def evaluate_erdos(h):
        h = np.array(h)
        h_shifted = 1 - h
        corr = np.correlate(h, h_shifted, mode='full')
        return np.max(corr) * dx
    
    # Genetic algorithm parameters
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    
    population = [copy.copy(initial_h) for _ in range(pop_size)]
    best_h = initial_h
    best_c5 = evaluate_erdos(best_h)
    
    for gen in range(generations):
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_erdos(h)
            fitness.append((c5, h))
        # Sort by fitness
        fitness.sort()
        # Select top half
        selected = [h for (c5, h) in fitness[:pop_size//2]]
        # Generate new population
        new_population = []
        while len(new_population) < pop_size:
            # Crossover
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            child = [0.5 * (p1 + p2) for p1, p2 in zip(parent1, parent2)]
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            # Normalize to ensure sum constraint
            total = sum(child)
            if total != 0:
                child = [x * (n_points / 2) / total for x in child]
            new_population.append(child)
        population = new_population
        # Update best
        for c5, h in fitness:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        # Early stopping if time is up
        if gen * (generations / budget_s) > 1:
