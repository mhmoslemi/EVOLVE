# Strategy: Use a greedy algorithm with simulated annealing to refine the initial solution
# by adjusting the values while maintaining the constraint sum(h) = n / 2

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Use 100 samples for better precision without excessive computation
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points]) / 20  # Scale down to fit the new n_points
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the constraint sum(h) = n / 2

    # Function to compute the overlap integral
    def compute_c5(h):
        return max(np.correlate(h, 1 - h, mode="full")) * dx

    # Simulated annealing
    T = 1.0  # Initial temperature
    T_min = 0.001
    cooling_rate = 0.995
    max_iterations = 1000
    best_h = np.copy(h)
    best_c5 = compute_c5(best_h)
    iterations = 0
    while T > T_min and iterations < max_iterations and time.time() - start_time < budget_s:
        for _ in range(100):
            # Randomly perturb one element
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            new_h = np.copy(h)
            new_h[idx] += delta
            new_h = np.clip(new_h, 0, 1)
            # Normalize to meet the constraint
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            c5 = compute_c5(new_h)
            # Accept with probability
            if np.random.rand() < np.exp((best_c5 - c5) / T):
                h = new_h
                if c5 < best_c5:
                    best_h = new_h
                    best_c5 = c5
        T *= cooling_rate
        iterations += 1

    return best_h, best_c5, n_points
