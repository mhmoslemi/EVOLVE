def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # fewer points for faster optimization
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values[:n_points])  # Crop initial solution
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)  # Normalize to meet the constraint

    def objective_func(h_values):
        h = h_values.reshape(n_points)
        # Compute the correlation
        corr = correlate(h, 1 - h, mode="full")
        # Compute the max value
        c5 = np.max(corr) * dx
        return c5

    def constraint_func(h_values):
        h = h_values.reshape(n_points)
        return np.sum(h) - n_points / 2  # Constraint: sum(h) == n_points / 2

    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    initial_guess = initial_h

    # Optimization
    start_time = time.time()
    result = minimize(objective_func, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_func}])
    elapsed_time = time.time() - start_time

    if elapsed_time > budget_s:
        return (initial_guess, objective_func(initial_guess), n_points)

    h_values = result.x
    c5_bound = objective_func(h_values)

    return (h_values, c5_bound, n_points)
