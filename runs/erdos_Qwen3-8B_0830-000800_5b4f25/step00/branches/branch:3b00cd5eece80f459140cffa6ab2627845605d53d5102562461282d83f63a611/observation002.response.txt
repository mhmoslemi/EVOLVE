def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for speed
    dx = 2.0 / n_points
    
    # Start with initial_h_values
    initial_h = np.array(initial_h_values[:n_points])
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    c5_initial = evaluate_erdos_solution(initial_h)
    best_c5 = c5_initial
    best_h = initial_h
    
    # Try with a few different n_points
    for n_points in [100, 200, 300, 400]:
        if time.time() - start_time > budget_s - 1:
            break
        dx = 2.0 / n_points
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        def objective(x):
            h = x.reshape(n_points)
            return evaluate_erdos_solution(h)
        
        def constraint(x):
            h = x.reshape(n_points)
            return np.sum(h) - (n_points / 2)
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})

            best_c5 = result.fun
            best_h = result.x
        
        print(f"n_points={n_points}, best_c5={best_c5}")
    
    # Try simulated annealing with the best solution found
    best_h = best_h if best_h is not None else initial_h
    best_h = best_h.reshape(n_points)
    T = 1.0
    cooling_rate = 0.995
    iterations = 1000
    for step in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
        T *= cooling_rate
        for i in range(n_points):
            new_h = copy.deepcopy(best_h)
            new_h[i] = np.random.uniform(0, 1)
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            c5_new = evaluate_erdos_solution(new_h)
            delta = c5_new - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                best_h = new_h
                best_c5 = c5_new
        print(f"Annealing step {step}, best_c5={best_c5}")
    
    return best_h.tolist(), best_c5, n_points
