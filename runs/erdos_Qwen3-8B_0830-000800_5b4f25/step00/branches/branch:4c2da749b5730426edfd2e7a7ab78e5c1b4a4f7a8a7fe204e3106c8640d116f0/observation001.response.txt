def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h /= np.sum(h)  # Ensure the integral is 1
    h = np.clip(h, 0, 1)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    def evaluate_h(h_values):
        h_values = np.array(h_values, dtype=np.float64)
        integral = np.sum(h_values) * (2.0 / n_points)
        if not np.isclose(integral, 1.0, atol=1e-6):
            return np.inf
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    def optimize_h(h_values, max_iter=100):
        h_values = np.array(h_values, dtype=np.float64)
        h_values = np.clip(h_values, 0, 1)
        best_c5 = evaluate_h(h_values)
        best_h = h_values.copy()
        
        for _ in range(max_iter):
            # Perturb the solution
            perturb = np.random.normal(0, 0.01, size=n_points)
            h_perturb = h_values + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            c5 = evaluate_h(h_perturb)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturb.copy()
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = optimize_h(h)
    
    while time.time() - start_time < budget_s:
        best_h, best_c5 = optimize_h(best_h)
    
    return best_h.tolist(), best_c5, n_points
