def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    conv = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5
