    np.random.seed(seed)
    random.seed(seed)

    # Normalize initial_h to meet the constraint sum(h) * dx = 1
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    # Define the objective function for minimization
    def objective(h):
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral C5
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraints: sum(h) * dx = 1 --> sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - (n_points / 2)
    
    # Setup optimization problem
    h_bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Perform optimization with a time budget
    start_time = time.time()
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=h_bounds, constraints=cons, tol=1e-6)
    
    # Get the best solution found
    best_h = result.x
    best_c5 = result.fun
    
    # Track time and return best solution found
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        print("Time budget exceeded, returning initial solution")
        return (initial_h, objective(initial_h), n_points)
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
