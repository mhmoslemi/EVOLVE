        perturbation = np.random.normal(0, step_size, n_points)
        new_h = initial_h + perturbation

        # Ensure integral constraint
        integral = np.sum(new_h) * dx

        new_c5 = compute_c5(new_h)

        if new_c5 < best_c5:
