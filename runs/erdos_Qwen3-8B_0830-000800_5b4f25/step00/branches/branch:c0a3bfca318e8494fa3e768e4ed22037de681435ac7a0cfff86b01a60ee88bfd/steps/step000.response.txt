import numpy as np
import scipy.optimize as opt
import time
import copy
import cvxpy as cp

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are pre-imported

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points for faster evaluation

    # Start from the initial solution
    h = copy.deepcopy(initial_h_values)
    n_points = len(h)
    h = np.array(h)

    # Normalize the initial solution
    h = h / np.sum(h) * (n_points / 2.0)

    def compute_c5(h_values):
        h_vals = h_values.reshape(-1)
        dx = 2.0 / len(h_vals)
        h = h_vals
        k_values = np.arange(-len(h) + 1, len(h))
        overlaps = []
        for k in k_values:
            if k < 0:
                shifted = np.roll(h, -k)
            else:
                shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted))
            overlaps.append(overlap)
        return max(overlap * dx for overlap in overlaps)

    def objective_func(h_vals):
        return compute_c5(h_vals)

    def constraint_func(h_vals):
        return np.sum(h_vals) - (n_points / 2.0)

    # Optimization using a gradient descent approach with random restarts
    for _ in range(10):  # 10 random restarts
        # Random perturbation
        perturbations = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = h + perturbations
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2.0)

        # Use a local optimizer to refine the perturbed solution
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(lambda x: compute_c5(x), x0=h_perturbed, bounds=bounds, method='L-BFGS-B', tol=1e-4, options={'maxiter': 100})
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                print(f"Found better solution: C5 = {best_c5}")
        
        # Check time budget
        if time.time() - start_time > budget_s - 5:
            break

    # Try a few more steps with a different starting point
    for _ in range(5):
        # Randomly generate a new starting point
        h_rand = np.random.uniform(0, 1, size=n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2.0)

        # Use a local optimizer to refine the perturbed solution
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(lambda x: compute_c5(x), x0=h_rand, bounds=bounds, method='L-BFGS-B', tol=1e-4, options={'maxiter': 100})
        if result.success:

            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                print(f"Found better solution: C5 = {best_c5}")
        
        # Check time budget
        if time.time() - start_time > budget_s - 5:
            break

    # Final check and return
    return (best_h.tolist(), best_c5, n_points)
