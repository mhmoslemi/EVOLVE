import numpy as np
import scipy.optimize as optimize
import time

# Import provided functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial construction using provided initial_h_values
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # Target C₅ bound
    target_c5 = 0.380876
    
    # Start with the initial construction
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try a few different starting points based on the initial construction
    for _ in range(3):  # Try 3 different starting points
        # Create perturbations of the initial construction
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points_initial)
        h_perturbed = np.clip(h_initial + perturbation, 0, 1)
        
        # Normalize the perturbed h to satisfy integral constraint
        total_integral = np.sum(h_perturbed) * (2.0 / n_points_initial)
        if total_integral != 0:
            h_perturbed = h_perturbed * (1.0 / total_integral) * (n_points_initial / 2.0)
        
        # Evaluate the perturbed h
        current_c5 = evaluate_erdos_solution(h_perturbed)
        
        # Update best solution if current is better
        if current_c5 < best_c5:
            best_h = h_perturbed
            best_c5 = current_c5
    
    # If we have a good enough solution, return early
    if best_c5 <= target_c5:
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            # Add some small random perturbations to the best solution
            perturbation = np.random.uniform(-0.01, 0.01, size=len(best_h))
            best_h = np.clip(best_h + perturbation, 0, 1)
            # Normalize again
            total_integral = np.sum(best_h) * (2.0 / len(best_h))
            if total_integral != 0:
                best_h = best_h * (1.0 / total_integral) * (len(best_h) / 2.0)
            best_c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), best_c5, len(best_h))
    
    # If the initial solution isn't good enough, try to increase resolution
    n_points = n_points_initial
    while n_points < 500 and time.time() - start_time < budget_s:
        n_points += 100
        # Create a new h with more samples
        h = np.zeros(n_points)
        # Use the previous best as a starting point
        h[:len(best_h)] = best_h
        # Fill in the rest with random values
        h[len(best_h):] = np.random.uniform(0, 1, size=n_points - len(best_h))
        # Normalize to satisfy integral constraint
        total_integral = np.sum(h) * (2.0 / n_points)
        if total_integral != 0:
            h = h * (1.0 / total_integral) * (n_points / 2.0)
        
        # Evaluate the new h
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution if current is better
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    # Final check for time budget
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Add some small random perturbations to the best solution
        perturbation = np.random.uniform(-0.01, 0.01, size=len(best_h))
        best_h = np.clip(best_h + perturbation, 0, 1)
        # Normalize again
        total_integral = np.sum(best_h) * (2.0 / len(best_h))
        if total_integral != 0:
            best_h = best_h * (1.0 / total_integral) * (len(best_h) / 2.0)
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, len(best_h))
