def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(initial_h_values)
    
    # Define the objective function
    def objective(h):
        # Ensure the area constraint
        h = np.clip(h, 0, 1)
        area = np.sum(h)
        if area < n_points / 2 - 1e-6 or area > n_points / 2 + 1e-6:
            return float('inf')
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        return c5
    
    # Define the bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization - use L-BFGS-B
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-5,
        options={'maxiter': 100}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    
    # Perform a simple grid search for neighborhood improvement
    perturbations = np.random.uniform(-0.1, 0.1, size=n_points)
    for _ in range(10):
        perturbed_h = best_h + perturbations
        perturbed_h = np.clip(perturbed_h, 0, 1)
        area = np.sum(perturbed_h)
        if area < n_points / 2 - 1e-6 or area > n_points / 2 + 1e-6:
            continue
        corr = np.correlate(perturbed_h, 1 - perturbed_h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        if c5 < best_c5:
            best_h = perturbed_h
            best_c5 = c5
    
    # Check if we have used the full budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Use a greedy hill-climbing approach
    for _ in range(50):
        if elapsed_time >= budget_s - 1:
            break
        # Try flipping a bit
        idx = np.random.randint(n_points)
        h = best_h.copy()
        h[idx] = 1 - h[idx]
        area = np.sum(h)
        if area < n_points / 2 - 1e-6 or area > n_points / 2 + 1e-6:
            continue
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        # Try small perturbation
        h = best_h.copy()
        h += np.random.uniform(-0.05, 0.05, size=n_points)
        h = np.clip(h, 0, 1)
        area = np.sum(h)
        if area < n_points / 2 - 1e-6 or area > n_points / 2 + 1e-6:
            continue
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return (best_h.tolist(), best_c5, n_points)
