import numpy as np
import scipy.optimize
import copy
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    total_area = sum(initial_h) * dx
    assert np.isclose(total_area, 1.0), "Initial h does not satisfy the area constraint"
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h_array = np.array(h)
        # Compute (1 - h) for the second signal
        one_minus_h = 1.0 - h_array
        # Compute the convolution (correlation)
        correlation = np.correlate(h_array, one_minus_h, mode="full")
        # Multiply by dx and take the maximum value
        c5 = np.max(correlation) * dx
        return c5
    
    # Function to perform gradient descent with momentum to minimize C5
    def optimize_h(h_initial, max_iterations=100, learning_rate=0.1, momentum=0.9, tolerance=1e-5):
        h = np.array(h_initial)
        velocity = np.zeros_like(h)
        best_h = copy.deepcopy(h)
        best_c5 = compute_c5(h)
        
        for i in range(max_iterations):
            # Compute current C5
            current_c5 = compute_c5(h)
            
            # Compute the gradient of C5 with respect to h
            # To approximate the gradient, we use finite differences
            gradient = np.zeros_like(h)
            for idx in range(len(h)):
                # Perturb h at idx
                h_plus = h.copy()
                h_plus[idx] += 1e-5
                c5_plus = compute_c5(h_plus)
                
                h_minus = h.copy()
                h_minus[idx] -= 1e-5
                c5_minus = compute_c5(h_minus)
                
                gradient[idx] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update velocity and h
            velocity = momentum * velocity - learning_rate * gradient
            h += velocity
            
            # Ensure h is in [0, 1]
            h = np.clip(h, 0.0, 1.0)
            
            # Ensure h satisfies the area constraint
            area = np.sum(h) * dx
            if not np.isclose(area, 1.0):
                # Normalize h to satisfy area constraint
                h = h / (area / 1.0)
            
            # Update best solution
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = copy.deepcopy(h)
            
            # Check for convergence
            if np.abs(current_c5 - best_c5) < tolerance:
                break
        
        return best_h, best_c5
    
    # Start optimization
    start_time = time.time()
    
    # Run optimization
    optimized_h, c5_bound = optimize_h(initial_h, max_iterations=500, learning_rate=0.1, momentum=0.9, tolerance=1e-5)
    
    # Compute the final c5 and n_points
    n_points = len(optimized_h)
    dx = 2.0 / n_points
    final_c5 = compute_c5(optimized_h)
    
    # Ensure the solution satisfies the area constraint
    area = np.sum(optimized_h) * dx
    if not np.isclose(area, 1.0):
        optimized_h = optimized_h / (area / 1.0)
        final_c5 = compute_c5(optimized_h)
    
    # Ensure we are within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time budget exceeded. Returning best solution so far.")
    
    return optimized_h, final_c5, n_points
