```python
import numpy as np
import scipy.optimize as optimize
import time
import copy
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 70  # Start with the initial n_points
    
    # Start with the initial solution
    h_values = copy.deepcopy(initial_h_values)
    
    # Define the time budget
    start_time = time.time()
    
    # Function to compute C5 from h_values
    def compute_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        one_minus_h = 1.0 - h
        # Compute the cross-correlation
        cross_corr = np.correlate(h, one_minus_h, mode="full")
        # The maximum correlation is the C5 value
        c5 = (np.max(cross_corr) * dx)
        return c5
    
    # Function to optimize h_values to minimize C5
    def optimize_h_values(h_initial, n_points):
        # Start with the initial h_values
        h = np.array(h_initial)
        
        # Ensure the sum of h equals n_points / 2
        total_sum = np.sum(h)
        adjustment_scalar = (n_points / 2 - total_sum) / n_points
        
        # Add a small random perturbation to avoid local optima
        h += np.random.normal(0, 0.05, size=n_points)
        h = np.clip(h, 0.0, 1.0)
        h += adjustment_scalar
        
        # Ensure the sum is correct
        h = np.clip(h, 0.0, 1.0)
        h = (h * (n_points / 2)) / np.sum(h)
        
        # Optimize using simulated annealing
        def objective_function(h_flat):
            h = h_flat.reshape((n_points,))
            total_sum = np.sum(h)
            h = (h * (n_points / 2)) / total_sum
            h = np.clip(h, 0.0, 1.0)
            cross_corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(cross_corr) * (2.0 / n_points)
            return c5
        
        # Use simulated annealing
        result = optimize.minimize(
            objective_function,
            x0=h.flatten(),
            method="anneal",
            options={'T': 1.0, 'tau': 0.95, 'maxiter': 1000}
        )
        
        # Reshape the result back to h
        optimized_h = result.x.reshape((n_points,))
        return optimized_h
    
    # Try different numbers of points
    for n_points in [70, 100, 150, 200, 250, 300, 350, 400]:
        # Check if we have exceeded the time budget
        if time.time() - start_time > budget_s - 10:
            break
        
        # Optimize h_values for current n_points
        optimized_h = optimize_h_values(h_values, n_points)
        
        # Ensure the sum is correct
        total_sum = np.sum(optimized_h)
        optimized_h = (optimized_h * (n_points / 2)) / total_sum
        optimized_h = np.clip(optimized_h, 0.0, 1.0)
        
        # Compute the C5 value
        c5 = compute_c5(optimized_h)
        
        # Print the result
        print(f"n_points = {n_points}, c5 = {c5}")
        
        # Update the best solution
        h_values = optimized_h
    
    # Compute the final C5 value
    final_c5 = compute_c5(h_values)
    
    # Return the best solution found
    return (h_values.tolist(), final_c5, len(h_values))
```
